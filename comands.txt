# 1. Base de datos
docker compose up -d

# 2. Backend
cd server
python -m venv .venv
source .venv/bin/activate        # Windows: .venv\Scripts\activate
pip install -r requirements.txt
cp .env.example .env
uvicorn app.main:app --reload --port 8080
# Docs: http://localhost:8080/docs

# 3. Frontend
cd client
npm install
cp .env.example .env
npm run tauri dev                # desarrollo
npm run tauri build              # produce .app / .msi / .AppImage
